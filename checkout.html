/* ============================================================ assets/cart-page.js — rendering voor cart.html ------------------------------------------------------------ Bevat GEEN eigen winkelwagen- of kortingslogica: alles komt uit window.velora*-functies die cart.js en discount.js al aanbieden. Deze module doet uitsluitend: lezen, op het scherm zetten, en bij interactie de bestaande functies aanroepen. Vereist: products.js, config.js, cart.js, discount.js — alle vóór dit bestand geladen (zie de scripttags in cart.html). ============================================================ */ (() => { 'use strict'; const fmt = window.veloraFmt; const container = document.getElementById('cartPageContent'); function render() { const cart = window.veloraGetCart(); if (!cart.length) { container.innerHTML = `

Je winkelwagen is nog leeg.

Ontdek producten
`; return; } const totals = window.veloraGetCartTotals(); const discountAmount = window.veloraCalculateDiscountAmount(totals.subtotal); const finalTotal = totals.subtotal - discountAmount; const applied = window.veloraGetAppliedDiscount(); container.innerHTML = ` ${cart .map( (item) => ` ` ) .join('')}
ProductAantalPrijsVerwijderen
${item.title}
${fmt(item.price * item.qty)}

${totals.freeShippingRemaining > 0 ? `Nog ${fmt(totals.freeShippingRemaining)} tot gratis verzending` : 'Je verzending is gratis!'}

${ applied ? `
Code ${applied.code} toegepast
` : `
` }
Subtotaal ${fmt(totals.subtotal)}
${ discountAmount > 0 ? `
Korting−${fmt(discountAmount)}
` : '' }
Verzending ${totals.freeShippingRemaining > 0 ? 'Berekend bij afrekenen' : 'Gratis'}
Totaal ${fmt(finalTotal)}
Afrekenen
${buildRecommendationsHTML(cart)} `; bindEvents(); } /* Aanbevolen producten / upsell — data-gegrond op wat er al in de winkelwagen ligt (via de bestaande veloraGetFrequentlyBoughtTogether/ veloraGetRelatedProducts, geen eigen aanbevelingslogica), producten die al in de wagen liggen worden uitgesloten. Hergebruikt de ene gedeelde kaart-template (veloraProductCardHTML) — geen duplicatie. */ function buildRecommendationsHTML(cart) { const cartIds = new Set(cart.map((i) => i.id)); const anchorProduct = window.VELORA_PRODUCTS.find((p) => p.id === cart[0].id); if (!anchorProduct) return ''; const combined = [ ...window.veloraGetFrequentlyBoughtTogether(anchorProduct, 4), ...window.veloraGetRelatedProducts(anchorProduct, 4), ].filter((p) => !cartIds.has(p.id)); const seen = new Set(); const recommendations = combined.filter((p) => (seen.has(p.id) ? false : seen.add(p.id))).slice(0, 4); if (!recommendations.length) return ''; return `
Compleet je bestelling

Vaak samen besteld

${recommendations.map((p) => window.veloraProductCardHTML(p, { showQuickAdd: false })).join('')}
`; } function bindEvents() { container.querySelectorAll('[data-qty-increase]').forEach((btn) => btn.addEventListener('click', () => { const id = Number(btn.dataset.qtyIncrease); const cart = window.veloraGetCart(); const item = cart.find((i) => i.id === id); if (item) window.veloraUpdateCartItemQty(id, item.qty + 1); render(); }) ); container.querySelectorAll('[data-qty-decrease]').forEach((btn) => btn.addEventListener('click', () => { const id = Number(btn.dataset.qtyDecrease); const cart = window.veloraGetCart(); const item = cart.find((i) => i.id === id); if (item) window.veloraUpdateCartItemQty(id, item.qty - 1); render(); }) ); container.querySelectorAll('[data-qty-input]').forEach((input) => input.addEventListener('change', () => { const id = Number(input.dataset.qtyInput); const qty = Math.max(1, Number(input.value) || 1); window.veloraUpdateCartItemQty(id, qty); render(); }) ); container.querySelectorAll('[data-remove]').forEach((btn) => btn.addEventListener('click', () => { window.veloraRemoveFromCart(Number(btn.dataset.remove)); render(); }) ); const discountForm = document.getElementById('discountForm'); discountForm?.addEventListener('submit', async (e) => { e.preventDefault(); const input = document.getElementById('discountInput'); const messageEl = document.getElementById('discountMessage'); const result = window.veloraApplyDiscountCode(input.value); if (result.success) { render(); return; } // Niet gevonden in de bestaande, statische lijst (WELKOM10/VELORA5)? // Dan controleren of het een dynamisch gegenereerde nieuwsbrief- // welkomstcode is (WELKOM10-XXXXXX) — volledig aanvullend, raakt // de bestaande synchrone flow hierboven niet aan. const typedCode = input.value.trim().toUpperCase(); if (typedCode.startsWith('WELKOM10-') || typedCode.startsWith('REWARDS-')) { try { const res = await fetch('/api/validate-discount-code', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ code: typedCode }), }); const data = await res.json(); if (data.valid) { window.veloraApplyDynamicDiscount(typedCode, data.type, data.value, data.label); render(); return; } if (messageEl) messageEl.innerHTML = `

${data.error}

`; return; } catch (err) { console.error('Kortingscode-validatie mislukt:', err); if (messageEl) messageEl.innerHTML = `

Kon de code niet controleren. Is er nog geen backend gekoppeld?

`; return; } } if (messageEl) messageEl.innerHTML = `

${result.message}

`; }); document.getElementById('removeDiscount')?.addEventListener('click', () => { window.veloraRemoveDiscountCode(); render(); }); } render(); })();
