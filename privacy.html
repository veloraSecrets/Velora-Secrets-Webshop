/* ============================================================ assets/orders-page.js — logica voor orders.html ------------------------------------------------------------ Haalt echte bestellingen op bij /api/orders (bestaat nog niet — net als /api/create-payment vóórdat Mollie gekoppeld was). Toont NOOIT verzonnen/voorbeeld-bestellingen: is het endpoint niet bereikbaar, dan zeg je dat expliciet i.p.v. nepdata te tonen. Vereist: auth.js, vóór dit bestand geladen. ============================================================ */ (() => { 'use strict'; const fmt = window.veloraFmt; (async () => { const user = await window.veloraRequireAuth(); if (!user) return; document.getElementById('accountLoading').hidden = true; document.getElementById('accountContent').hidden = false; const listEl = document.getElementById('ordersList'); try { const res = await fetch('/api/orders', { credentials: 'same-origin' }); if (!res.ok) throw new Error(`Server gaf status ${res.status}`); const { orders } = await res.json(); if (!orders || !orders.length) { listEl.innerHTML = `

Je hebt nog geen bestellingen geplaatst.

Ontdek producten
`; return; } listEl.innerHTML = orders .map( (order) => `
Bestelling ${order.id} ${order.statusLabel || order.status}
${order.date} · ${fmt(order.total)}
` ) .join(''); } catch (err) { console.error('Kon bestelgeschiedenis niet laden:', err); listEl.innerHTML = `

Kon je bestelgeschiedenis niet laden. Is er nog geen backend gekoppeld aan /api/orders?

`; } })(); })();
